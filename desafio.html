class DadoProtegido:
    def __init__(self, texto: str):
        self._texto = texto
    def exibir_dado_protegido(self):
        print(f"Dado: '{self._texto}'")

class Usuario(DadoProtegido):
    def __init__(self, nome: str, texto_senha: str):
        super().__init__(texto_senha)
        self.nome = nome
    def descobrir_senha(self) -> str:
        return self._texto[::-1]
